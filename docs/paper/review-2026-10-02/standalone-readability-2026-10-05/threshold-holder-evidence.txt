	if n.Direct == nil {
		return func() {}, nil
	}
	policy, err := n.Direct.Policy(n.Schedule, n.Organizations)
	if err != nil {
		return nil, err
	}
	raw, err := os.ReadFile(c.RepairKeyFile)
	if err != nil {
		return nil, err
	}
	signer, err := chameleon.DecodeSigner(policy.Key, raw)
	if err != nil {
		return nil, err
	}

		return nil, protocol.ErrAuth
	}
